def check_secret_rw(env_vars, mount_path):
    key = "ami_validation_{0}".format(utc_now_str())
    value = "ok"

    run_cmd(["vault", "kv", "put", mount_path, "{0}={1}".format(key, value)], env=env_vars)
    secret_data = run_json_cmd(["vault", "kv", "get", "-format=json", mount_path], env=env_vars)

    data_block = secret_data.get("data", {})
    if isinstance(data_block.get("data"), dict):
        actual = data_block.get("data", {}).get(key)   # KV v2
        kv_version = "v2"
    else:
        actual = data_block.get(key)                   # KV v1
        kv_version = "v1"

    if actual != value:
        raise ValidationError(
            "Secret write/read mismatch on KV {0}. Expected {1}, got {2}".format(
                kv_version, value, actual
            )
        )

    return CheckResult(
        "Secret read/write",
        "PASS",
        "Temporary validation key was written and read successfully.",
        "Functional",
        {
            "path": mount_path,
            "key": key,
            "value": actual,
            "kv_version_detected": kv_version
        }
    )
